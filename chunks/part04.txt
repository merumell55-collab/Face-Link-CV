function displayFromSource(point, metrics, view, mirrored=false) {
return sourceToCanvas(point, metrics, view, mirrored);
}
function faceScale(pts) {
if (!pts || pts.length < 4) return 1;
const eyeL = { x:(pts[0].x + pts[1].x)/2, y:(pts[0].y + pts[1].y)/2 };
const eyeR = { x:(pts[2].x + pts[3].x)/2, y:(pts[2].y + pts[3].y)/2 };
return Math.max(1, Math.hypot(eyeL.x - eyeR.x, eyeL.y - eyeR.y));
}
function circumcircle(a,b,c) {
const d = 2 * (a.x*(b.y-c.y) + b.x*(c.y-a.y) + c.x*(a.y-b.y));
if (Math.abs(d) < 1e-9) return null;
const a2 = a.x*a.x + a.y*a.y, b2 = b.x*b.x + b.y*b.y, c2 = c.x*c.x + c.y*c.y;
const ux = (a2*(b.y-c.y) + b2*(c.y-a.y) + c2*(a.y-b.y)) / d;
const uy = (a2*(c.x-b.x) + b2*(a.x-c.x) + c2*(b.x-a.x)) / d;
return { x:ux, y:uy, r2:(ux-a.x)**2 + (uy-a.y)**2 };
}
function delaunay(points) {
if (points.length < 3) return [];
let minX=Infinity,minY=Infinity,maxX=-Infinity,maxY=-Infinity;
points.forEach(p => { minX=Math.min(minX,p.x); minY=Math.min(minY,p.y); maxX=Math.max(maxX,p.x); maxY=Math.max(maxY,p.y); });
const d = Math.max(maxX-minX, maxY-minY);
const mx = (minX+maxX)/2, my = (minY+maxY)/2;
const pts = points.map(p => ({...p}));
const s0 = pts.length, s1 = s0+1, s2 = s0+2;
pts.push({x:mx - 20*d, y:my - d}, {x:mx, y:my + 20*d}, {x:mx + 20*d, y:my - d});
let tris = [[s0,s1,s2]];
for (let i=0;i<points.length;i++) {
const bad = [];
for (let t=0;t<tris.length;t++) {
const tri = tris[t];
const cc = circumcircle(pts[tri[0]], pts[tri[1]], pts[tri[2]]);
if (cc && ((pts[i].x-cc.x)**2 + (pts[i].y-cc.y)**2 <= cc.r2 + 1e-6)) bad.push(t);
}

for (const idx of bad) {
const tr = tris[idx];
[[tr[0],tr[1]],[tr[1],tr[2]],[tr[2],tr[0]]].forEach(e => {
const rev = edges.findIndex(x => x[0]===e[1] && x[1]===e[0]);
if (rev >= 0) edges.splice(rev, 1);
else edges.push(e);
});
}
tris = tris.filter((_, idx) => !bad.includes(idx));
edges.forEach(e => tris.push([e[0], e[1], i]));
}
