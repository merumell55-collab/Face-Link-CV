umber(els.smooth.value) / 100);
state.smoothedCamPoints = state.smoothedCamPoints.map((p, i) => ({
x: p.x + (state.currentCamPoints[i].x - p.x) * alpha,
y: p.y + (state.currentCamPoints[i].y - p.y) * alpha
}));
renderPhotoWarp();
renderCamBase();
state.gray.copyTo(state.prevGray);
state.nextPts.copyTo(state.prevPts);
} catch (e) {
console.error(e);
stopTracking();
setStatus("追跡中にエラーが発生しました。ポイントを再設定してやり直してください。", "bad");
return;
}
state.trackRaf = requestAnimationFrame(trackLoop);
}
